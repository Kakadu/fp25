Module Lambda_lib.Ast

type name = string
type 'name t =
  1. | Var of 'name
    (*

    Variable x

    *)
  2. | Fun of 'name * 'name t
    (*

    fun *

    *)
  3. | App of 'name t * 'name t
    (*

    Application f g

    *)
  4. | Int of int
  5. | Neg of 'name t
  6. | Bin of binop * 'name t * 'name t
  7. | If of 'name t * 'name t * 'name t
  8. | Let of 'name * 'name t * 'name t
  9. | LetRec of 'name * 'name t * 'name t
