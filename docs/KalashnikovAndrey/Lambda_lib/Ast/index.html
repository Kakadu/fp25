Module Lambda_lib.Ast

type op =
  1. | Add
    (*

    Addition operator +

    *)
  2. | Sub
    (*

    Subtraction operator -

    *)
  3. | Mul
    (*

    Multiplication operator *

    *)
  4. | Div
    (*

    Division operator /

    *)
  5. | Lt
    (*

    Less-than comparison <

    *)
  6. | Eq
    (*

    Equality comparison =

    *)
  7. | Mt
    (*

    Greater-than comparison >

    *)
type rec_flag =
  1. | Rec
    (*

    Recursive let rec binding

    *)
  2. | Val
    (*

    Non-recursive let binding

    *)
type expr =
  1. | Const of int
    (*

    Integer literal

    *)
  2. | BinOp of op * expr * expr
    (*

    Binary operation with two operands

    *)
  3. | Var of string
    (*

    Variable

    *)
  4. | Let of rec_flag * string * expr * expr
    (*

    let or let rec binding

    *)
  5. | If of expr * expr * expr
    (*

    Conditional if then else expression

    *)
  6. | App of expr * expr
    (*

    Function application

    *)
  7. | Fun of string * expr
    (*

    Lambda abstraction with one argument

    *)
  8. | Neg of expr
    (*

    Unary minus

    *)
